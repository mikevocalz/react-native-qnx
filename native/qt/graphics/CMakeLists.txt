cmake_minimum_required(VERSION 3.31.9)
project(RnqnxQtGraphics LANGUAGES CXX)

if(NOT DEFINED RNQNX_ENABLE_QT_GRAPHICS)
  option(RNQNX_ENABLE_QT_GRAPHICS "Build the experimental public Qt graphics surface observer" OFF)
endif()
if(NOT RNQNX_ENABLE_QT_GRAPHICS)
  return()
endif()
if(NOT CMAKE_SYSTEM_NAME MATCHES "^(Linux|Darwin|QNX)$")
  message(FATAL_ERROR "RNQNX_ENABLE_QT_GRAPHICS requires a Linux, macOS or QNX Qt host")
endif()
find_package(Qt6 6.11.1 REQUIRED COMPONENTS Core Gui Quick)
add_library(rnqnx_qt_graphics STATIC
  GraphicsSurfaceState.cpp GraphicsSurfaceState.h QtGraphicsSurface.cpp QtGraphicsSurface.h)
set_target_properties(rnqnx_qt_graphics PROPERTIES AUTOMOC ON POSITION_INDEPENDENT_CODE ON)
target_compile_features(rnqnx_qt_graphics PUBLIC cxx_std_20)
target_compile_definitions(rnqnx_qt_graphics PRIVATE QT_NO_KEYWORDS)
target_include_directories(rnqnx_qt_graphics PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(rnqnx_qt_graphics PUBLIC Qt6::Core Qt6::Gui Qt6::Quick)
