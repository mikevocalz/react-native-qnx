cmake_minimum_required(VERSION 3.31.9)
project(RnqnxLifecycleObserver LANGUAGES CXX)
option(RNQNX_BUILD_LIFECYCLE_OBSERVER "Build the isolated native lifecycle observer" OFF)
option(RNQNX_LIFECYCLE_SANITIZERS "Instrument observer and tests with ASan/UBSan" OFF)
if(NOT RNQNX_BUILD_LIFECYCLE_OBSERVER)
  return()
endif()
if(NOT CMAKE_SYSTEM_NAME MATCHES "^(Linux|QNX)$")
  message(FATAL_ERROR "This experiment supports Linux and QNX build targets only")
endif()
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.11.1 REQUIRED COMPONENTS Core Gui)
add_library(rnqnx_lifecycle_observer STATIC QtApplicationLifecycle.cpp QtApplicationLifecycle.h)
target_compile_features(rnqnx_lifecycle_observer PUBLIC cxx_std_17)
target_include_directories(rnqnx_lifecycle_observer PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(rnqnx_lifecycle_observer PUBLIC Qt6::Core Qt6::Gui)
if(RNQNX_LIFECYCLE_SANITIZERS)
  if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    message(FATAL_ERROR "Sanitizers here are qualified for Linux Clang/GNU only")
  endif()
  target_compile_options(rnqnx_lifecycle_observer PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(rnqnx_lifecycle_observer PUBLIC -fsanitize=address,undefined)
endif()
include(CTest)
if(BUILD_TESTING)
  find_package(Qt6 6.11.1 REQUIRED COMPONENTS Test)
  find_package(Threads REQUIRED)
  add_executable(rnqnx_lifecycle_test LifecycleTest.cpp)
  target_link_libraries(rnqnx_lifecycle_test PRIVATE rnqnx_lifecycle_observer Qt6::Test Threads::Threads)
  add_test(NAME rnqnx_lifecycle COMMAND rnqnx_lifecycle_test)
  set_tests_properties(rnqnx_lifecycle PROPERTIES TIMEOUT 30 ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
