cmake_minimum_required(VERSION 3.31.9)
project(RnqnxApplicationIdentity LANGUAGES CXX)
option(RNQNX_BUILD_APP_IDENTITY_TESTS "Build isolated application identity tests" OFF)
option(RNQNX_APP_IDENTITY_SANITIZERS "Instrument the identity tests" OFF)
if(NOT RNQNX_BUILD_APP_IDENTITY_TESTS)
  return()
endif()
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "The isolated identity test lane is Linux-only")
endif()
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.11.1 REQUIRED COMPONENTS Core Gui Test)
add_executable(rnqnx_app_identity_test ApplicationIdentityTest.cpp)
target_compile_features(rnqnx_app_identity_test PRIVATE cxx_std_17)
target_include_directories(rnqnx_app_identity_test PRIVATE "../../vnext/include")
target_link_libraries(rnqnx_app_identity_test PRIVATE Qt6::Core Qt6::Gui Qt6::Test)
if(RNQNX_APP_IDENTITY_SANITIZERS)
  if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    message(FATAL_ERROR "Identity sanitizers require Clang or GNU")
  endif()
  target_compile_options(rnqnx_app_identity_test PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(rnqnx_app_identity_test PRIVATE -fsanitize=address,undefined)
endif()
include(CTest)
add_test(NAME rnqnx_app_identity COMMAND rnqnx_app_identity_test)
set_tests_properties(rnqnx_app_identity PROPERTIES TIMEOUT 30 ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
