cmake_minimum_required(VERSION 3.31.9)
project(RnqnxApplicationIdentity LANGUAGES CXX)
option(RNQNX_BUILD_APP_IDENTITY_TESTS "Build isolated application identity tests" OFF)
option(RNQNX_APP_IDENTITY_SANITIZERS "Instrument the identity tests" OFF)
option(RNQNX_BUILD_IDENTITY_TOOL "Build the host icon/label tool used by prepare-native" OFF)
if(NOT RNQNX_BUILD_APP_IDENTITY_TESTS AND NOT RNQNX_BUILD_IDENTITY_TOOL)
  return()
endif()
# A host tool and host tests: never part of a QNX or Marathon target image.
if(NOT CMAKE_SYSTEM_NAME MATCHES "^(Linux|Darwin)$" OR CMAKE_CROSSCOMPILING)
  message(FATAL_ERROR "The identity tool and tests build only for a Linux or macOS host")
endif()
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.11.1 REQUIRED COMPONENTS Core Gui)

add_library(rnqnx_identity_core STATIC IconSet.cpp LabelFit.cpp)
target_compile_features(rnqnx_identity_core PUBLIC cxx_std_17)
target_include_directories(rnqnx_identity_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" "../../vnext/include")
target_link_libraries(rnqnx_identity_core PUBLIC Qt6::Core Qt6::Gui)
# Same rule as the renderer: no bare signals/slots/emit, so this code can be
# shared with targets that define QT_NO_KEYWORDS.
target_compile_definitions(rnqnx_identity_core PUBLIC QT_NO_KEYWORDS)
set_target_properties(rnqnx_identity_core PROPERTIES AUTOMOC OFF)

add_executable(rnqnx_identity_tool IdentityTool.cpp)
target_link_libraries(rnqnx_identity_tool PRIVATE rnqnx_identity_core)
set_target_properties(rnqnx_identity_tool PROPERTIES AUTOMOC OFF)

if(NOT RNQNX_BUILD_APP_IDENTITY_TESTS)
  return()
endif()
find_package(Qt6 6.11.1 REQUIRED COMPONENTS Test)
add_executable(rnqnx_app_identity_test ApplicationIdentityTest.cpp)
add_executable(rnqnx_icon_set_test IconSetTest.cpp)
add_executable(rnqnx_label_fit_test LabelFitTest.cpp)
target_compile_definitions(rnqnx_label_fit_test PRIVATE
  RNQNX_IDENTITY_FONT_DIR="${CMAKE_CURRENT_SOURCE_DIR}/fonts")
set(identity_tests rnqnx_app_identity_test rnqnx_icon_set_test rnqnx_label_fit_test)
foreach(test_target IN LISTS identity_tests)
  target_compile_features(${test_target} PRIVATE cxx_std_17)
  target_link_libraries(${test_target} PRIVATE rnqnx_identity_core Qt6::Test)
endforeach()
if(RNQNX_APP_IDENTITY_SANITIZERS)
  if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    message(FATAL_ERROR "Identity sanitizers require Clang or GNU")
  endif()
  foreach(target_name IN LISTS identity_tests ITEMS rnqnx_identity_core rnqnx_identity_tool)
    target_compile_options(${target_name} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(${target_name} PRIVATE -fsanitize=address,undefined)
  endforeach()
endif()
include(CTest)
add_test(NAME rnqnx_app_identity COMMAND rnqnx_app_identity_test)
add_test(NAME rnqnx_icon_set COMMAND rnqnx_icon_set_test)
add_test(NAME rnqnx_label_fit COMMAND rnqnx_label_fit_test)
set_tests_properties(rnqnx_app_identity rnqnx_icon_set rnqnx_label_fit PROPERTIES
  TIMEOUT 60 ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
