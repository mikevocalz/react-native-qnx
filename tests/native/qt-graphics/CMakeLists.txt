cmake_minimum_required(VERSION 3.31.9)
project(RnqnxQtGraphicsTests LANGUAGES CXX)

if(NOT DEFINED RNQNX_ENABLE_QT_GRAPHICS)
  option(RNQNX_ENABLE_QT_GRAPHICS "Build the experimental public Qt graphics surface observer" OFF)
endif()
option(RNQNX_QT_GRAPHICS_SANITIZERS "Instrument the graphics observer and tests with ASan/UBSan" OFF)
if(NOT RNQNX_ENABLE_QT_GRAPHICS)
  return()
endif()
set(root "${CMAKE_CURRENT_LIST_DIR}/../../..")
if(NOT TARGET rnqnx_qt_graphics)
  add_subdirectory("${root}/native/qt/graphics" qt-graphics)
endif()
find_package(Qt6 6.11.1 REQUIRED COMPONENTS Test)
find_package(Threads REQUIRED)
add_executable(rnqnx_qt_graphics_test GraphicsSurfaceTest.cpp)
set_target_properties(rnqnx_qt_graphics_test PROPERTIES AUTOMOC ON)
target_compile_definitions(rnqnx_qt_graphics_test PRIVATE QT_NO_KEYWORDS)
target_link_libraries(rnqnx_qt_graphics_test PRIVATE rnqnx_qt_graphics Qt6::Test Threads::Threads)
foreach(target rnqnx_qt_graphics rnqnx_qt_graphics_test)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
endforeach()
if(RNQNX_QT_GRAPHICS_SANITIZERS)
  if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    message(FATAL_ERROR "This sanitizer lane requires Linux Clang/GNU")
  endif()
  foreach(target rnqnx_qt_graphics rnqnx_qt_graphics_test)
    target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-sanitize-recover=undefined -fno-omit-frame-pointer)
    target_link_options(${target} PRIVATE -fsanitize=address,undefined)
  endforeach()
endif()
include(CTest)
if(BUILD_TESTING AND NOT CMAKE_CROSSCOMPILING)
  add_test(NAME rnqnx_qt_graphics COMMAND rnqnx_qt_graphics_test -platform offscreen)
  set_tests_properties(rnqnx_qt_graphics PROPERTIES TIMEOUT 25
    ENVIRONMENT "QT_QUICK_BACKEND=software;QT_SCALE_FACTOR=1.25;ASAN_OPTIONS=detect_leaks=0:halt_on_error=1;UBSAN_OPTIONS=halt_on_error=1")
  add_test(NAME rnqnx_qt_graphics_null COMMAND rnqnx_qt_graphics_test -platform offscreen)
  set_tests_properties(rnqnx_qt_graphics_null PROPERTIES TIMEOUT 25
    ENVIRONMENT "QT_QUICK_BACKEND=;QSG_RHI_BACKEND=null;QT_SCALE_FACTOR=1.25;ASAN_OPTIONS=detect_leaks=0:halt_on_error=1;UBSAN_OPTIONS=halt_on_error=1")
endif()
