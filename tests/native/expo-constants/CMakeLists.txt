cmake_minimum_required(VERSION 3.31.9)
project(RnqnxExpoConstantsTests LANGUAGES C CXX)
include(CTest)
if(CMAKE_CROSSCOMPILING OR NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "Expo Constants host tests require native Linux")
endif()
get_filename_component(root "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
set(RNQNX_REACT_NATIVE_DIR "${root}/node_modules/react-native" CACHE PATH "Pinned RN source")
find_package(Qt6 6.11.1 REQUIRED COMPONENTS Core Gui)
if(NOT TARGET rnqnx_hermes)
  include("${root}/cmake/dependencies/Hermes.cmake")
endif()
if(NOT TARGET rnqnx_expo_modules_qnx)
  add_subdirectory("${root}/packages/expo-modules-core-qnx" expo-modules)
endif()
add_executable(expo_constants_runtime_test RuntimeTest.cpp)
target_compile_definitions(expo_constants_runtime_test PRIVATE RNQNX_TEST_CONFIGURED=$<BOOL:${RNQNX_EXPO_PUBLIC_CONFIG}>)
target_link_libraries(expo_constants_runtime_test PRIVATE rnqnx_expo_modules_qnx rnqnx_hermes Qt6::Core Qt6::Gui)
add_test(NAME expo_constants_runtime COMMAND expo_constants_runtime_test)
set_tests_properties(expo_constants_runtime PROPERTIES TIMEOUT 30 ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
if(RNQNX_ENABLE_EXPO_CONSTANTS)
  find_program(NODE_EXECUTABLE node REQUIRED)
  set(fixture "${CMAKE_CURRENT_BINARY_DIR}/official-expo-constants.js")
  add_custom_command(OUTPUT "${fixture}"
    COMMAND "${NODE_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/build-public-fixture.mjs" "${fixture}"
    DEPENDS "${CMAKE_CURRENT_LIST_DIR}/build-public-fixture.mjs" "${root}/pnpm-lock.yaml"
      "${root}/node_modules/expo-constants/src/Constants.ts" VERBATIM)
  add_custom_target(expo_constants_public_fixture DEPENDS "${fixture}")
  add_dependencies(expo_constants_runtime_test expo_constants_public_fixture)
  add_test(NAME expo_constants_public_sdk COMMAND expo_constants_runtime_test "${fixture}")
  set_tests_properties(expo_constants_public_sdk PROPERTIES TIMEOUT 30 ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
