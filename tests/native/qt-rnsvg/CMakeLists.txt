cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(RnqnxQtRnsvgTests LANGUAGES CXX)
  option(RNQNX_ENABLE_QT_RNSVG "Build react-native-svg element rendering" OFF)
  include(CTest)
endif()

if(NOT RNQNX_ENABLE_QT_RNSVG)
  return()
endif()

option(RNQNX_QT_RNSVG_SANITIZERS "Instrument the react-native-svg renderer and tests with ASan/UBSan" OFF)
find_package(Qt6 6.11.1 REQUIRED COMPONENTS Core Gui Quick Test)
if(NOT TARGET rnqnx_qt_rnsvg)
  add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../../../native/qt/rnsvg" "${CMAKE_CURRENT_BINARY_DIR}/rnsvg")
endif()

if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  target_compile_options(rnqnx_qt_rnsvg PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
if(RNQNX_QT_RNSVG_SANITIZERS)
  if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    message(FATAL_ERROR "react-native-svg sanitizers are qualified only for Linux Clang/GNU")
  endif()
  target_compile_options(rnqnx_qt_rnsvg PRIVATE -fsanitize=address,undefined -fno-sanitize-recover=undefined -fno-omit-frame-pointer)
endif()

# Every Rnsvg*Test.cpp here is one test executable. RnsvgRenderTest keeps its
# original name; the others are rnqnx_qt_rnsvg_<stem>_test, with the stem in
# lower case (RnsvgFilterTest.cpp -> rnqnx_qt_rnsvg_filter_test).
set(RNQNX_RNSVG_TEST_DATA "${CMAKE_CURRENT_LIST_DIR}")
file(GLOB RNQNX_RNSVG_TEST_SOURCES CONFIGURE_DEPENDS "${CMAKE_CURRENT_LIST_DIR}/Rnsvg*Test.cpp")
foreach(source IN LISTS RNQNX_RNSVG_TEST_SOURCES)
  get_filename_component(stem "${source}" NAME_WE)
  if(stem STREQUAL "RnsvgRenderTest")
    set(target rnqnx_qt_rnsvg_test)
  else()
    string(REGEX REPLACE "^Rnsvg(.*)Test$" "\\1" middle "${stem}")
    string(TOLOWER "${middle}" middle)
    set(target "rnqnx_qt_rnsvg_${middle}_test")
  endif()
  add_executable(${target} "${source}")
  set_target_properties(${target} PROPERTIES AUTOMOC ON)
  target_compile_features(${target} PRIVATE cxx_std_17)
  target_compile_definitions(${target} PRIVATE QT_NO_KEYWORDS
    RNQNX_RNSVG_TEST_DATA_DIR="${RNQNX_RNSVG_TEST_DATA}")
  target_link_libraries(${target} PRIVATE rnqnx_qt_rnsvg Qt6::Test)
  if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
  if(RNQNX_QT_RNSVG_SANITIZERS)
    target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-sanitize-recover=undefined -fno-omit-frame-pointer)
    target_link_options(${target} PRIVATE -fsanitize=address,undefined)
  endif()
  if(BUILD_TESTING AND NOT CMAKE_CROSSCOMPILING)
    add_test(NAME ${target} COMMAND ${target} -platform offscreen)
    set_tests_properties(${target} PROPERTIES TIMEOUT 120
      ENVIRONMENT "QT_QUICK_BACKEND=software;ASAN_OPTIONS=detect_leaks=0:halt_on_error=1;UBSAN_OPTIONS=halt_on_error=1")
  endif()
endforeach()
