cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(RnqnxQtSvgTests LANGUAGES CXX)
  option(RNQNX_ENABLE_QT_SVG "Build the experimental static Qt SVG component" OFF)
  include(CTest)
endif()

if(NOT RNQNX_ENABLE_QT_SVG)
  return()
endif()

option(RNQNX_QT_SVG_SANITIZERS "Instrument the Qt SVG source and tests with ASan/UBSan" OFF)
find_package(Qt6 6.11.1 REQUIRED COMPONENTS Core Gui Quick Svg Test)
if(NOT TARGET rnqnx_qt_svg)
  add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../../../native/qt/svg" "${CMAKE_CURRENT_BINARY_DIR}/svg")
endif()

add_executable(rnqnx_qt_svg_test SvgDocumentTest.cpp)
set_target_properties(rnqnx_qt_svg_test PROPERTIES AUTOMOC ON)
target_compile_features(rnqnx_qt_svg_test PRIVATE cxx_std_17)
target_compile_definitions(rnqnx_qt_svg_test PRIVATE QT_NO_KEYWORDS)
target_link_libraries(rnqnx_qt_svg_test PRIVATE rnqnx_qt_svg Qt6::Test)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  target_compile_options(rnqnx_qt_svg PRIVATE -Wall -Wextra -Wpedantic -Werror)
  target_compile_options(rnqnx_qt_svg_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
if(RNQNX_QT_SVG_SANITIZERS)
  if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    message(FATAL_ERROR "Qt SVG sanitizers are qualified only for Linux Clang/GNU")
  endif()
  foreach(target rnqnx_qt_svg rnqnx_qt_svg_test)
    target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-sanitize-recover=undefined -fno-omit-frame-pointer)
  endforeach()
  target_link_options(rnqnx_qt_svg_test PRIVATE -fsanitize=address,undefined)
endif()

if(BUILD_TESTING AND NOT CMAKE_CROSSCOMPILING)
  add_test(NAME rnqnx_qt_svg_test COMMAND rnqnx_qt_svg_test -platform offscreen)
  set_tests_properties(rnqnx_qt_svg_test PROPERTIES TIMEOUT 30
    ENVIRONMENT "QT_QUICK_BACKEND=software;ASAN_OPTIONS=detect_leaks=0:halt_on_error=1;UBSAN_OPTIONS=halt_on_error=1")
endif()
