cmake_minimum_required(VERSION 3.20)
project(rnqnx-text-input-bridge LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
find_package(Qt6 6.11.1 REQUIRED COMPONENTS Core Gui Qml Quick)
set(root "${CMAKE_CURRENT_LIST_DIR}/../../..")
add_executable(rnqnx_text_input_bridge_test BridgeTest.cpp
  "${root}/vnext/src/fabric/QtTextInputBridge.cpp"
  "${root}/vnext/include/react-native-qnx/fabric/QtTextInputBridge.h"
  "${root}/vnext/src/views/qt/QtTextInputQml.cpp")
set_target_properties(rnqnx_text_input_bridge_test PROPERTIES AUTOMOC ON)
target_include_directories(rnqnx_text_input_bridge_test PRIVATE "${root}/vnext/include")
target_compile_definitions(rnqnx_text_input_bridge_test PRIVATE QT_NO_KEYWORDS RNQNX_ENABLE_TEXT_INPUT_CONTRACT=1)
target_link_libraries(rnqnx_text_input_bridge_test PRIVATE Qt6::Core Qt6::Gui Qt6::Qml Qt6::Quick)
target_compile_options(rnqnx_text_input_bridge_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "QNX")
  target_compile_options(rnqnx_text_input_bridge_test PRIVATE -fsanitize=address,undefined -fno-sanitize-recover=undefined -fno-omit-frame-pointer)
  target_link_options(rnqnx_text_input_bridge_test PRIVATE -fsanitize=address,undefined)
endif()
if(BUILD_TESTING AND NOT CMAKE_CROSSCOMPILING)
  add_test(NAME rnqnx_text_input_bridge_test COMMAND rnqnx_text_input_bridge_test -platform offscreen)
  set_tests_properties(rnqnx_text_input_bridge_test PROPERTIES TIMEOUT 15 ENVIRONMENT "QT_QUICK_BACKEND=software;ASAN_OPTIONS=detect_leaks=0:halt_on_error=1;UBSAN_OPTIONS=halt_on_error=1")
endif()
