cmake_minimum_required(VERSION 3.20)
project(rnqnx-accessibility-view LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(CTest)
find_package(Qt6 6.11.1 REQUIRED COMPONENTS Core Gui Qml Quick)
if(NOT QT_FEATURE_accessibility)
 message(FATAL_ERROR "Accessibility View test requires Qt accessibility support")
endif()
set(root "${CMAKE_CURRENT_LIST_DIR}/../../..")
add_executable(rnqnx_accessibility_view_test ViewTest.cpp "${root}/vnext/src/fabric/QtViewAccessibility.cpp")
target_include_directories(rnqnx_accessibility_view_test PRIVATE "${root}/vnext/include")
target_compile_definitions(rnqnx_accessibility_view_test PRIVATE QT_NO_KEYWORDS RNQNX_ENABLE_ACCESSIBILITY_BRIDGE=1)
target_compile_options(rnqnx_accessibility_view_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "QNX")
 target_compile_options(rnqnx_accessibility_view_test PRIVATE -fsanitize=address,undefined -fno-sanitize-recover=undefined -fno-omit-frame-pointer)
 target_link_options(rnqnx_accessibility_view_test PRIVATE -fsanitize=address,undefined)
endif()
target_link_libraries(rnqnx_accessibility_view_test PRIVATE Qt6::Core Qt6::Gui Qt6::Qml Qt6::Quick)
if(BUILD_TESTING AND NOT CMAKE_CROSSCOMPILING)
 add_test(NAME rnqnx_accessibility_view_test COMMAND rnqnx_accessibility_view_test -platform offscreen)
 set_tests_properties(rnqnx_accessibility_view_test PROPERTIES TIMEOUT 15 ENVIRONMENT "QT_QUICK_BACKEND=software;ASAN_OPTIONS=detect_leaks=0:halt_on_error=1")
endif()
