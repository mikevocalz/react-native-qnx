cmake_minimum_required(VERSION 3.24)
project(rnqnx_qt_motion_tests LANGUAGES CXX)
include(CTest)

option(RNQNX_MOTION_SANITIZERS "Instrument motion library and its native test" OFF)
option(RNQNX_ENABLE_QT_MOTION "Build experimental native Qt motion" OFF)
if(NOT RNQNX_ENABLE_QT_MOTION)
  message(FATAL_ERROR "Motion tests require explicit RNQNX_ENABLE_QT_MOTION=ON")
endif()
if(NOT TARGET rnqnx_qt_motion)
  add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../../../native/qt/motion" motion)
endif()
add_executable(rnqnx_qt_motion_test MotionTest.cpp)
target_link_libraries(rnqnx_qt_motion_test PRIVATE rnqnx_qt_motion)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  target_compile_options(rnqnx_qt_motion_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
  if(RNQNX_MOTION_SANITIZERS AND NOT CMAKE_CROSSCOMPILING)
    foreach(target rnqnx_qt_motion rnqnx_qt_motion_test)
      target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-sanitize-recover=undefined -fno-omit-frame-pointer)
      target_link_options(${target} PUBLIC -fsanitize=address,undefined)
    endforeach()
  endif()
endif()
if(BUILD_TESTING AND NOT CMAKE_CROSSCOMPILING)
  add_test(NAME rnqnx_qt_motion_test COMMAND rnqnx_qt_motion_test -platform offscreen)
  set_tests_properties(rnqnx_qt_motion_test PROPERTIES TIMEOUT 15 ENVIRONMENT
    "QT_QUICK_BACKEND=software;ASAN_OPTIONS=detect_leaks=0:halt_on_error=1;UBSAN_OPTIONS=halt_on_error=1")
endif()
